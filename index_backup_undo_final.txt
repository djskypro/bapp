  const dbRequest = indexedDB.open("BappMediaDB", 1);
  dbRequest.onupgradeneeded = function(e) {
      let db = e.target.result;
      if (!db.objectStoreNames.contains("images")) {

  function saveImgToDB(id, base64) {
      return new Promise((resolve) => {
          if(!bappDB) {
              setTimeout(() => resolve(saveImgToDB(id, base64)), 100);
              return;
          }
          try {
              let tx = bappDB.transaction("images", "readwrite");
              let req = tx.objectStore("images").put(base64, id);
              req.onsuccess = function() { resolve(true); };
              req.onerror = function(e) { console.log("DB Save Error", e); resolve(false); };
          } catch(e) { console.log("DB Save Error", e); resolve(false); }
      });
  }
  
  function getImgFromDB(id, callback) {
      if(!bappDB) { 
          setTimeout(() => getImgFromDB(id, callback), 100);
          return; 
      }
      try {
          let tx = bappDB.transaction("images", "readonly");
          let req = tx.objectStore("images").get(id);
          req.onsuccess = function() { callback(req.result); };
          req.onerror = function() { callback(null); };
      } catch(e) { callback(null); }
  }

  function reconnectWebSocket() {
    reconnectTimer = null;

    // Save the current message handler so we don't lose it
    const currentHandler = socket.onmessage;
    
    // Create a fresh socket

        let queue = JSON.parse(localStorage.getItem('bapp_offline_queue')) || [];

            localStorage.removeItem('bapp_offline_queue');
        }
        socket.send(JSON.stringify({ type: 'register_user', username: myBappId, profile: myProfile }));
    };
    
    socket.onclose = () => {
        console.log("🔴 Disconnected again. Retrying...");
        if (!reconnectTimer) reconnectTimer = setTimeout(reconnectWebSocket, 3000);
    };
    
    socket.onerror = (err) => console.error("WebSocket Error: ", err);
    
    // Reattach all the message listeners we built earlier
    if (currentHandler) {
        socket.onmessage = currentHandler;
    }

      }
      
      playerDiv.appendChild(playBtn);
      playerDiv.appendChild(waveDiv);
      playerDiv.appendChild(timeSpan);
      
      // Hidden real audio element
      const audio = document.createElement('audio');
      audio.src = text;
      audio.style.display = 'none';
      
      playerDiv.appendChild(audio);
      newMsg.appendChild(playerDiv);
      
      // --- Player Logic ---
      let isPlaying = false;
      playBtn.onclick = function() {
        if (isPlaying) {
          audio.pause();
          playBtn.innerText = '▶';
          isPlaying = false;
        } else {
          audio.play();
          playBtn.innerText = '⏸';
          playBtn.classList.add('played'); // Turn blue
          isPlaying = true;
          
          // Mark all bars as played
          bars.forEach(b => b.classList.add('played'));
        }
      };
      
      audio.onended = function() {
        playBtn.innerText = '▶';
        isPlaying = false;
        // Reset bars to start if replayed later (optional)
      };
      
      audio.ontimeupdate = function() {
        const mins = Math.floor(audio.currentTime / 60);
        const secs = Math.floor(audio.currentTime % 60);
        timeSpan.innerText = mins + ':' + secs.toString().padStart(2, '0');
        
        // Calculate progress to color bars
        const progress = audio.currentTime / (audio.duration || 1);
        const playedBars = Math.floor(progress * numBars);
        bars.forEach((b, i) => {
          if (i < playedBars) b.classList.add('played');
          else if (audio.currentTime === 0) b.classList.remove('played');
        });
      };
    } else {
      // --- MATRIX DECRYPTION EFFECT ---
      if (type === 'received' && text.length > 0) {
        const textSpan = document.createElement('span');
        newMsg.appendChild(textSpan);
        
        const originalText = Array.from(text); // Handles emojis correctly
        const chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789!@#$%^&*()_+{}[]<>';
        let iteration = 0;
        
        textSpan.style.color = 'var(--primary)'; // Make it green during decryption
        textSpan.style.fontFamily = 'monospace';  // Hacker font

        const interval = setInterval(() => {
          textSpan.innerText = originalText.map((char, index) => {
            if (index < iteration) {
              return originalText[index]; // Revealed characters
            }
            if (char === ' ') return ' '; // Keep spaces as spaces
            return chars[Math.floor(Math.random() * chars.length)]; // Scrambled character
          }).join('');
          
          if (iteration >= originalText.length) {
            clearInterval(interval);
            textSpan.style.color = ''; // Reset to normal text color
            textSpan.style.fontFamily = ''; // Reset to normal font
          }
          iteration += 1 / 4; // Controls the speed of decryption (lower = slower)
        }, 30);
      } else {
        // Normal text for sent messages
        
      if (type === 'received' && typeof text === 'string' && text.length > 0 && !text.startsWith('data:')) {
          const textSpan = document.createElement('span');
          newMsg.appendChild(textSpan);
          
          const originalText = Array.from(text);
          const chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789!@#$%^&*()_+{}[]<>';
          let iteration = 0;
          
          textSpan.style.color = '#25D366';
          textSpan.style.fontFamily = 'monospace';
          textSpan.style.textShadow = '0 0 8px #25D366';
          
          const interval = setInterval(() => {
              textSpan.innerText = originalText.map((char, index) => {
                  if (index < iteration) return originalText[index];
                  if (char === ' ') return ' ';
                  return chars[Math.floor(Math.random() * chars.length)];
              }).join('');
              
              if (iteration >= originalText.length) {
                  clearInterval(interval);
                  textSpan.style.color = '';
                  textSpan.style.fontFamily = '';
                  textSpan.style.textShadow = '';
              }
              iteration += 1 / 3;
          }, 30);
      } else {
          newMsg.appendChild(document.createTextNode(text));
      }

      }
    }

    const timeSpan = document.createElement('span');
    timeSpan.className = 'msg-time';
    timeSpan.textContent = timeStr || new Date().toLocaleTimeString([], {hour: '2-digit', minute:'2-digit'});
    newMsg.appendChild(timeSpan);

    if (type === 'sent') {
      const check = document.createElement('span');
      check.className = 'msg-status ' + (status === 'read' ? 'read' : '');
      check.innerText = status === 'pending' ? '🕒' : ((status === 'read' || status === 'delivered') ? '✓✓' : '✓');
      newMsg.appendChild(check);
    }
    
    // Insert before the spacer if it exists, otherwise just append
    // --- FLY ANIMATION FOR SENT MESSAGES ---
    if (type === 'sent') {
        newMsg.classList.add('materialize-in');
        setTimeout(() => newMsg.classList.remove('materialize-in'), 500); 
    }

    // --- WATER DROP SPLASH ANIMATION ---
    if (type === 'sent') {
        newMsg.classList.add('msg-water-pop');
        setTimeout(() => newMsg.classList.remove('msg-water-pop'), 550); 
    }

    const spacer = document.getElementById('chat-spacer');
    if (spacer) {
        msgBox.insertBefore(newMsg, spacer);
    } else {
        msgBox.appendChild(newMsg);
    }
    msgBox.scrollTop = msgBox.scrollHeight;
  }

  function sendMessage() {
    const input = document.getElementById('chat-input');
    const text = input.value.trim();
    if (text === "") return;

    const to = activeChatId;
    const reply = currentReplyTo;
    const now = new Date().toLocaleTimeString([], {hour: '2-digit', minute:'2-digit'});
    const msgId = Date.now().toString();
    
    const isConnected = socket.readyState === WebSocket.OPEN;
    const status = isConnected ? 'sent' : 'pending';

    // 1. Show on screen immediately
    appendMessage(text, 'sent', now, status, false, false, msgId, reply, false, false, myProfile.bubbleColor || '#005c4b');
      
      
    input.value = '';
    triggerInputDrop();
 
    // Snap the keyboard back open instantly after sending
    setTimeout(() => {
        const chatIn = document.getElementById('chat-input');
        const chatView = document.getElementById('chat-view');
        if (chatIn && chatView && chatView.classList.contains('active')) {
            chatIn.focus();
        }
    }, 10);

    // 2. Save to local memory
    try {
        const savedMsgs = JSON.parse(localStorage.getItem('bapp_msgs_' + to)) || [];
        savedMsgs.push({ text: text, type: 'sent', time: now, status: status, msgId: msgId, replyData: reply, bubbleColor: myProfile.bubbleColor || '#005c4b' });
        localStorage.setItem('bapp_msgs_' + to, JSON.stringify(savedMsgs));
        loadChatList();
    } catch(e) { console.error("Storage full"); }

    const rp = document.getElementById('reply-preview');
    if (rp) rp.style.display = 'none';
    currentReplyTo = null;
    if (window.replyState && activeChatId) delete window.replyState[activeChatId];

    // 3. Send now OR save to Offline Queue
    const payload = {
        type: 'send_message',
        to: to,
        from: myBappId,
        text: text,
        msgId: msgId,
        replyData: reply,
        bubbleColor: myProfile.bubbleColor || '#005c4b'
    };

    if (isConnected) {
        socket.send(JSON.stringify(payload));
    } else {
        let queue = JSON.parse(localStorage.getItem('bapp_offline_queue')) || [];
        queue.push(payload);
        localStorage.setItem('bapp_offline_queue', JSON.stringify(queue));
        console.log("📩 Message queued. Will send when online.");
    }
  }

  function handleImageUpload(event) {
    const file = event.target.files[0];
    if(!file) return;
    
    event.target.value = ''; 

    // Show modal
    document.getElementById('media-preview-modal').style.display = 'flex';
    document.getElementById('preview-loading').style.display = 'block';
    document.getElementById('preview-content').innerHTML = '';
    const sendBtn = document.getElementById('preview-send-btn');
    sendBtn.disabled = true;
    sendBtn.style.background = '#333';
    sendBtn.style.color = '#777';

    const reader = new FileReader();
    reader.onload = function(e) {
      if (file.type.startsWith('video/')) {
         pendingIsVideo = true;
         pendingMediaBase64 = e.target.result;
         
         document.getElementById('preview-loading').style.display = 'none';
         document.getElementById('preview-content').innerHTML = '<video src="' + pendingMediaBase64 + '" controls style="max-width:100%; max-height:70vh; border-radius:15px;"></video>';
         
         sendBtn.disabled = false;
         sendBtn.style.background = '#25D366';
         sendBtn.style.color = 'black';
      } else {
         pendingIsVideo = false;
         const img = new Image();
         img.onload = function() {
           const canvas = document.createElement('canvas');
           const MAX_WIDTH = 1080; const MAX_HEIGHT = 1080;
           let width = img.width; let height = img.height;
           if (width > height) { if (width > MAX_WIDTH) { height *= MAX_WIDTH / width; width = MAX_WIDTH; } }
           else { if (height > MAX_HEIGHT) { width *= MAX_HEIGHT / height; height = MAX_HEIGHT; } }
           canvas.width = width; canvas.height = height;
           const ctx = canvas.getContext('2d');
           ctx.drawImage(img, 0, 0, width, height);
           pendingMediaBase64 = canvas.toDataURL('image/jpeg', 0.7);
           
           document.getElementById('preview-loading').style.display = 'none';
           document.getElementById('preview-content').innerHTML = '<img src="' + pendingMediaBase64 + '" style="max-width:100%; max-height:70vh; border-radius:15px;">';
           
           sendBtn.disabled = false;
           sendBtn.style.background = '#25D366';
           sendBtn.style.color = 'black';
         };
         img.src = e.target.result;
      }
    };
    reader.readAsDataURL(file);
  }

  function confirmMediaSend() {
    if(!pendingMediaBase64) return;
    
    // Instantly close modal
    document.getElementById('media-preview-modal').style.display = 'none';
    
    const to = activeChatId;
    const base64Img = pendingMediaBase64;
    const isVid = pendingIsVideo;
    const msgId = Date.now().toString();
    const reply = currentReplyTo;
    
    const payload = { 
        type: 'send_message', 
        to: to, 
        from: myBappId, 
        text: base64Img, 
        isImage: !isVid, 
        isVideo: isVid, 
        msgId: msgId, 
        replyData: reply 
    };
    
    // CHECK IF ONLINE, IF NOT, QUEUE IT
    if (socket.readyState === WebSocket.OPEN) {
        socket.send(JSON.stringify(payload));
    } else {
        let queue = JSON.parse(localStorage.getItem('bapp_offline_queue')) || [];
        queue.push(payload);
        localStorage.setItem('bapp_offline_queue', JSON.stringify(queue));
        console.log("📩 Image queued. Will send when online.");
    }
    const now = new Date().toLocaleTimeString([], {hour: '2-digit', minute:'2-digit'});
    appendMessage(base64Img, 'sent', now, 'sent', !isVid, false, msgId, reply, false, isVid);

    const rp = document.getElementById('reply-preview');
    if (rp) rp.style.display = 'none';
    currentReplyTo = null;
    if (window.replyState && activeChatId) delete window.replyState[activeChatId];

    try {
        const savedMsgs = JSON.parse(localStorage.getItem('bapp_msgs_' + to)) || [];
        savedMsgs.push({ text: base64Img, type: 'sent', time: now, isImage: !isVid, isVideo: isVid, status: 'sent' });
        localStorage.setItem('bapp_msgs_' + to, JSON.stringify(savedMsgs));
        loadChatList();

        document.getElementById('reply-preview').style.display = 'none';
        currentReplyTo = null;
        if (window.replyState) delete window.replyState[to];
    } catch(err) { 
        // Auto-cleanup
        for (let i = 0; i < localStorage.length; i++) {
            let key = localStorage.key(i);
            if (key.startsWith('bapp_msgs_')) {
                let msgs = JSON.parse(localStorage.getItem(key)) || [];
                let changed = false;
                msgs.forEach(m => {
                    if (m.isImage && m.text && m.text.startsWith('data:image')) {
                        m.text = "📷 [Image removed to save space]";
                        m.isImage = false;
                        changed = true;
                    }
                });
                if (changed) localStorage.setItem(key, JSON.stringify(msgs));
            }
        }
        try {
            const savedMsgs = JSON.parse(localStorage.getItem('bapp_msgs_' + to)) || [];
            savedMsgs.push({ text: base64Img, type: 'sent', time: now, isImage: !isVid, isVideo: isVid, status: 'sent' });
            localStorage.setItem('bapp_msgs_' + to, JSON.stringify(savedMsgs));
            loadChatList();
        } catch(e2) { alert("Phone storage is completely full."); }
    }
    
    pendingMediaBase64 = null;
  }

            sub.style.fontWeight = 'normal';
          }
        }

        // 2. Update the Chat List preview
        const previewEl = document.getElementById('preview-' + senderId);
        if (previewEl) {
          if (data.isRecording) {
            previewEl.innerText = '🔴 recording audio...';
            previewEl.style.color = '#e74c3c';
            previewEl.style.fontWeight = 'bold';
          } else {
            // Revert to last message
            const msgs = JSON.parse(localStorage.getItem('bapp_msgs_' + senderId)) || [];
            let lastMsgText = 'Tap to chat';
            if (msgs.length > 0) {

        if (activeChatId === senderId) {

        }

        // 2. Update the Chat List on the Home Screen
        const previewEl = document.getElementById('preview-' + senderId);
        if (previewEl) {
          if (data.isTyping) {

        msgs.forEach(m => { if(m.type === 'sent') m.status = 'read'; });

  window.MediaRecorder = function(stream, options) {
    const recorder = new originalMediaRecorder(stream, options);
    
    recorder.addEventListener('start', () => {
      const canvas = document.getElementById('live-waveform');
      if (!canvas) return;
      
      canvas.style.display = 'block';
      const canvasCtx = canvas.getContext('2d');
      canvas.width = canvas.clientWidth;
      canvas.height = canvas.clientHeight;
      
      const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
      const source = audioCtx.createMediaStreamSource(stream);
      const analyser = audioCtx.createAnalyser();
      analyser.fftSize = 64;
      source.connect(analyser);
      
      function draw() {
        if (!canvas.style.display || canvas.style.display === 'none') return;
        requestAnimationFrame(draw);
        
        const bufferLength = analyser.frequencyBinCount;
        const dataArray = new Uint8Array(bufferLength);
        analyser.getByteFrequencyData(dataArray);
        
        canvasCtx.clearRect(0, 0, canvas.width, canvas.height);
        canvasCtx.fillStyle = '#25D366'; // Green live waves
        
        const barWidth = (canvas.width / bufferLength) * 1.5;
        let x = 0;
        for(let i = 0; i < bufferLength; i++) {
            const barHeight = (dataArray[i] / 255) * canvas.height;
            canvasCtx.fillRect(x, canvas.height - barHeight, barWidth, barHeight);
            x += barWidth + 2;
        }
      }
      draw();
    });

    recorder.addEventListener('stop', () => {
      const canvas = document.getElementById('live-waveform');
      if (canvas) canvas.style.display = 'none';
    });

    return recorder;
  };

    const input = document.createElement('input');

    input.accept = 'image/*,video/*';
    input.multiple = true; // Allow multiple files
    
    input.onchange = async (e) => {

            continue;

  function processFile(file) {
    return new Promise((resolve) => {
      const reader = new FileReader();
      reader.onload = (event) => {
        const base64Data = event.target.result;
        if (file.type.startsWith('image/')) {
            const img = new Image();
            img.onload = () => {
              const canvas = document.createElement('canvas');
              canvas.width = 600; canvas.height = 600 * (img.height / img.width);
              const ctx = canvas.getContext('2d');
              ctx.drawImage(img, 0, 0, canvas.width, canvas.height);
              resolve(canvas.toDataURL('image/jpeg', 0.6));
            };
            img.src = base64Data;
        } else {
            resolve(base64Data); // Video
        }
      };
      reader.readAsDataURL(file);
    });
  }

<script>
  // --- RIGHT EDGE BACK GESTURE ---
  let touchStartX = 0;
  let touchStartY = 0;
  let isEdgeSwipe = false;
  let isSwipingBack = false;
  
  // Increase this number to make the edge easier to grab
  const EDGE_WIDTH = 50; 
  const SWIPE_THRESHOLD = 60;

  document.addEventListener('touchstart', (e) => {
    // Ignore if we are interacting with a message bubble
    if (e.target.closest('.msg')) return;
    
    const screenWidth = window.innerWidth;
    const touchX = e.touches[0].clientX;

    // Check if the touch started within EDGE_WIDTH of the RIGHT edge
    if (touchX > screenWidth - EDGE_WIDTH) {
      touchStartX = touchX;
      touchStartY = e.touches[0].clientY;
      isEdgeSwipe = true;
      isSwipingBack = false;
    } else {
      isEdgeSwipe = false;
    }
  }, { passive: false });

  document.addEventListener('touchmove', (e) => {
    if (!isEdgeSwipe) return;
    
    const deltaX = e.touches[0].clientX - touchStartX;
    const deltaY = e.touches[0].clientY - touchStartY;

    // If moving mostly horizontally to the LEFT
    if (deltaX < -10 && Math.abs(deltaY) < Math.abs(deltaX)) {
      isSwipingBack = true;
      // Prevent the browser's default back gesture
      if (e.cancelable) e.preventDefault(); 
    } else if (Math.abs(deltaY) > Math.abs(deltaX)) {
      isEdgeSwipe = false;
    }
  }, { passive: false });

  document.addEventListener('touchend', (e) => {
    if (!isEdgeSwipe) return;
    
    const deltaX = e.changedTouches[0].clientX - touchStartX;
    
    // If swiped LEFT by more than the threshold, trigger the back action
    if (isSwipingBack && deltaX < -SWIPE_THRESHOLD) {
      handleBackGesture();
    }
    
    isEdgeSwipe = false;
    isSwipingBack = false;
  });

  function handleBackGesture() {

  // --- BREATHING BACKGROUND (Canvas) ---
  const canvas = document.getElementById('bg-canvas');
  const ctx = canvas.getContext('2d');
  let dots = [];
  const SPACING = 25; // Matched from your CSS
  const RADIUS = 8;   // Matched from your CSS gradient size

  function resizeCanvas() {
    canvas.width = window.innerWidth;
    canvas.height = window.innerHeight;
    initDots();
  }

  function initDots() {
    dots = [];
    const centerX = canvas.width / 2;
    const centerY = canvas.height / 2;
    const maxDist = 300; // Matched your 600px width (radius 300)

    for (let x = 0; x < canvas.width; x += SPACING) {
      for (let y = 0; y < canvas.height; y += SPACING) {
        const dist = Math.hypot(x - centerX, y - centerY);
        
        if (dist < maxDist) {
            // Smooth fade at the edges (like your mask-image)
            let maskAlpha = 1 - (dist / maxDist);
            maskAlpha = Math.pow(maskAlpha, 1.2);

            if (maskAlpha > 0.05) {
                // Base alpha matched your CSS (0.04) but slightly boosted for canvas
                let baseAlpha = 0.08 * maskAlpha; 
                
                // Check if light mode
                const isLight = document.body.classList.contains('light-mode');
                
                dots.push({
                  x: x,
                  y: y,
                  baseAlpha: baseAlpha,
                  currentAlpha: baseAlpha,
                  state: 'idle',
                  speed: 0,
                  waitTime: 0,
                  color: isLight ? '0,0,0' : '255,255,255'
                });
            }
        }
      }
    }
  }

  function animateBackground() {
    requestAnimationFrame(animateBackground);
    ctx.clearRect(0, 0, canvas.width, canvas.height);

    // Randomly pick a dot to start fading out
    if (Math.random() > 0.90) { // Controls how often dots blink
      const randomDot = dots[Math.floor(Math.random() * dots.length)];
      if (randomDot && randomDot.state === 'idle') {
        randomDot.state = 'fadingOut';
        randomDot.speed = Math.random() * 0.01 + 0.005; // Random fade speed
      }
    }

    // Update and draw dots
    dots.forEach(dot => {
      if (dot.state === 'fadingOut') {
        dot.currentAlpha -= dot.speed;
        if (dot.currentAlpha <= 0) {
          dot.currentAlpha = 0;
          dot.state = 'waitingToFadeIn';
          dot.waitTime = Date.now() + (Math.random() * 3000 + 500); // Wait 0.5 to 3.5 sec
        }
      } else if (dot.state === 'waitingToFadeIn') {
        if (Date.now() > dot.waitTime) {
          dot.state = 'fadingIn';
        }
      } else if (dot.state === 'fadingIn') {
        dot.currentAlpha += dot.speed;
        if (dot.currentAlpha >= dot.baseAlpha) {
          dot.currentAlpha = dot.baseAlpha;
          dot.state = 'idle';
        }
      }

      // Draw the ring (hollow circle)
      ctx.beginPath();
      ctx.arc(dot.x, dot.y, RADIUS, 0, Math.PI * 2);
      ctx.strokeStyle = `rgba(${dot.color}, ${dot.currentAlpha})`;
      ctx.lineWidth = 3;
      ctx.stroke();
    });
  }

    function processBappCommand(command) {
      // Pattern 1: Send Message (e.g., "message kevin I am coming")
      const messageMatch = command.match(/(?:message|text|tell)\s+([a-z0-9_]+)\s+(.*)/i);
      // Pattern 2: Call (e.g., "call kevin")
      const callMatch = command.match(/(?:call|phone)\s+([a-z0-9_]+)/i);

      if (messageMatch) {
        const targetName = messageMatch[1].trim();
        const msgText = messageMatch[2].trim();
        executeBappAction(targetName, 'message', msgText);
      } 
      else if (callMatch) {
        const targetName = callMatch[1].trim();
        executeBappAction(targetName, 'call');
      } 
      else {
        speakBappAI("I didn't understand. Try saying 'Message Alex hello' or 'Call Alex'.");
      }
    }

    function executeBappAction(targetName, actionType, messageText) {
      // Find the user in saved chats or profiles
      const profiles = JSON.parse(localStorage.getItem('bapp_profiles')) || {};
      let foundUserId = null;
      let foundDisplayName = null;

      // Search through profiles
      for (let id in profiles) {
        if (id.toLowerCase() === targetName || (profiles[id].name && profiles[id].name.toLowerCase().includes(targetName))) {
          foundUserId = id;
          foundDisplayName = profiles[id].name || id;
          break;
        }
      }

      if (!foundUserId) {
        // Fallback: Check local chats
        const chats = JSON.parse(localStorage.getItem('bapp_chats')) || [];
        for (let id of chats) {
          if (id.toLowerCase() === targetName) {
            foundUserId = id;
            foundDisplayName = id;
            break;
          }
        }
      }

      if (!foundUserId) {
        speakBappAI("I couldn't find " + targetName + " in your contacts.");
        return;
      }

      // Execute the action
      if (actionType === 'message') {
        speakBappAI("Sending message to " + foundDisplayName);
        openChat(foundUserId);
        setTimeout(() => {
            const input = document.getElementById('chat-input');
            input.value = messageText;
            
            // TRIGGER THE INPUT EVENT so the UI knows text was added
            input.dispatchEvent(new Event('input', { bubbles: true }));
            
            // Wait a tiny bit for the UI to catch up, then send
            setTimeout(() => {
                sendMessage();
            }, 150);
        }, 500); // Wait for chat to open
      } 
      else if (actionType === 'call') {
        speakBappAI("Calling " + foundDisplayName);
        openChat(foundUserId);
        setTimeout(() => {
            startCall('audio');
        }, 800);
      }
    }
